<script>

let tracking = false;
let startTime = null;
let timerInterval = null;
let movementScore = 0;
let motionEvents = 0;
let confidence = 0;
let lastMovement = Date.now();

function requestPermissions() {

  if (typeof DeviceMotionEvent !== 'undefined' && typeof DeviceMotionEvent.requestPermission === 'function') {

    DeviceMotionEvent.requestPermission()
      .then(permissionState => {
        if (permissionState === 'granted') {
          alert('Motion permission granted');
        } else {
          alert('Motion permission denied');
        }
      })
      .catch(console.error);

  } else {
    alert('Motion permissions not required on this device/browser.');
  }
}

function startTracking() {

  tracking = true;
  startTime = Date.now();
  movementScore = 0;
  motionEvents = 0;
  confidence = 0;

  document.getElementById('statusBox').innerHTML = '<strong>Status:</strong> Indoor activity tracking active.';

  document.getElementById('motionDot').classList.add('active');

  timerInterval = setInterval(updateTimer, 1000);
}

function stopTracking() {

  tracking = false;

  clearInterval(timerInterval);

  document.getElementById('statusBox').innerHTML = '<strong>Status:</strong> Tracking stopped.';

  document.getElementById('motionDot').classList.remove('active');

  if (confidence >= 70 && motionEvents > 40) {
    document.getElementById('validationDot').classList.add('active');
    document.getElementById('statusBox').innerHTML = '<strong>Status:</strong> Valid indoor activity session detected.';
  }
}

function updateTimer() {

  if (!tracking) return;

  const elapsed = Math.floor((Date.now() - startTime) / 1000);

  const mins = String(Math.floor(elapsed / 60)).padStart(2, '0');
  const secs = String(elapsed % 60).padStart(2, '0');

  document.getElementById('timer').innerText = `${mins}:${secs}`;
}

window.addEventListener('devicemotion', function(event) {

  if (!tracking) return;

  const acc = event.accelerationIncludingGravity;

  if (!acc) return;

  document.getElementById('accelerometerDot').classList.add('active');

  const x = acc.x || 0;
  const y = acc.y || 0;
  const z = acc.z || 0;

  const magnitude = Math.sqrt(x * x + y * y + z * z);

  document.getElementById('liveData').innerHTML = `
    X: ${x.toFixed(2)}<br>
    Y: ${y.toFixed(2)}<br>
    Z: ${z.toFixed(2)}<br>
    Motion Magnitude: ${magnitude.toFixed(2)}
  `;

  if (magnitude > 13) {

    movementScore += Math.floor(magnitude);
    motionEvents++;
    lastMovement = Date.now();

    document.getElementById('movementScore').innerText = movementScore;
    document.getElementById('motionEvents').innerText = motionEvents;

    confidence = Math.min(100, Math.floor((motionEvents / 50) * 100));

    document.getElementById('confidence').innerText = confidence + '%';
    document.getElementById('confidenceFill').style.width = confidence + '%';
  }

  if (Date.now() - lastMovement > 5000) {
    confidence = Math.max(0, confidence - 1);
    document.getElementById('confidence').innerText = confidence + '%';
    document.getElementById('confidenceFill').style.width = confidence + '%';
  }

});

window.addEventListener('deviceorientation', function(event) {

  if (!tracking) return;

  document.getElementById('gyroDot').classList.add('active');

});

</script>
